Citadele Integration — 3 Lines

Drop NexusJS into any page. Pega CDH enrichment begins immediately.

citadele-nexus.js 
<!-- NexusJS SDK — DCS CDH Bridge Integration -->
<script src="https://cdn.nexuscdp.io/v1/nexus.js"></script>

// Initialise for Citadele
NexusCDP.init({ apiKey: 'pk_citadele_prod', bank: 'Citadele' });

// Auto-maps to Pega CDH Interaction History schema
NexusCDP.track('product_view', { product: 'consumer-loan', rate: '5.9%' });
NexusCDP.identify('CUST-LV-001', { name: 'Anete Kalniņa', bank: 'Citadele' });
